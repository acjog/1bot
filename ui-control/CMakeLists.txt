cmake_minimum_required(VERSION 3.16)
project(OneBotControl VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(NOT WIN32)
  message(FATAL_ERROR "The control panel currently requires Windows (DPAPI)")
endif()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets Network Concurrent LinguistTools)
set(UI_SOURCES main.cpp wiring.cpp updater.cpp modelspage.cpp pluginspage.cpp accesscard.cpp localization.cpp)
set(UI_LANGUAGES zh_CN en_US ja_JP ko_KR zh_TW)
set(UI_TS_FILES)
foreach(language IN LISTS UI_LANGUAGES)
  list(APPEND UI_TS_FILES "${CMAKE_CURRENT_SOURCE_DIR}/${language}.ts")
endforeach()
qt5_add_translation(UI_QM_FILES ${UI_TS_FILES})
add_custom_target(update_translations
    COMMAND Qt5::lupdate -no-obsolete ${UI_SOURCES} -ts ${UI_TS_FILES}
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    VERBATIM)
add_custom_target(release_translations DEPENDS ${UI_QM_FILES})
get_target_property(QT_QMAKE Qt5::qmake IMPORTED_LOCATION)
execute_process(COMMAND "${QT_QMAKE}" -query QT_INSTALL_TRANSLATIONS
    OUTPUT_VARIABLE QT_TRANSLATIONS_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
set(UI_QRC "${CMAKE_CURRENT_BINARY_DIR}/translations.qrc")
file(WRITE "${UI_QRC}" "<RCC><qresource prefix=\"/i18n\">\n")
foreach(qm IN LISTS UI_QM_FILES)
  get_filename_component(qm_name "${qm}" NAME)
  file(APPEND "${UI_QRC}" "<file alias=\"${qm_name}\">${qm}</file>\n")
endforeach()
foreach(language IN LISTS UI_LANGUAGES)
  set(qt_catalog "${QT_TRANSLATIONS_DIR}/qtbase_${language}.qm")
  if(NOT EXISTS "${qt_catalog}")
    string(REGEX REPLACE "_.*" "" base_language "${language}")
    set(qt_catalog "${QT_TRANSLATIONS_DIR}/qtbase_${base_language}.qm")
  endif()
  if(EXISTS "${qt_catalog}")
    file(APPEND "${UI_QRC}" "<file alias=\"qtbase_${language}.qm\">${qt_catalog}</file>\n")
  endif()
endforeach()
file(APPEND "${UI_QRC}" "</qresource></RCC>\n")
qt5_add_resources(UI_TRANSLATION_RESOURCES "${UI_QRC}")
add_executable(1bot-control WIN32 ${UI_SOURCES} ${UI_TRANSLATION_RESOURCES})
add_dependencies(1bot-control release_translations)
install(FILES ${UI_QM_FILES} DESTINATION translations)
set(PPOCR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../ppocr" CACHE PATH "Existing ppocr dependencies")
set(CURL_ROOT "${PPOCR_ROOT}/lib/3rd_lib/win32/release/curl")
set(OPENSSL_RUNTIME_DIR "${PPOCR_ROOT}/lib/autoupdater" CACHE PATH "OpenSSL 1.1 x86 DLL directory")
set(UPDATE_CA_BUNDLE "${PPOCR_ROOT}/win32/curl-ca-bundle.crt" CACHE FILEPATH "Trusted CA bundle")
target_include_directories(1bot-control PRIVATE "${CURL_ROOT}/include")
target_link_libraries(1bot-control PRIVATE Qt5::Widgets Qt5::Network Qt5::Concurrent
    "${CURL_ROOT}/lib/curl_x86.lib" Crypt32 ws2_32 wldap32)
set(UPDATE_RUNTIME_FILES
    "${CURL_ROOT}/bin/libcurl.dll"
    "${OPENSSL_RUNTIME_DIR}/libssl-1_1.dll"
    "${OPENSSL_RUNTIME_DIR}/libcrypto-1_1.dll")
foreach(runtime IN LISTS UPDATE_RUNTIME_FILES)
  if(NOT EXISTS "${runtime}")
    message(FATAL_ERROR "Missing HTTPS runtime: ${runtime}")
  endif()
endforeach()
if(NOT EXISTS "${UPDATE_CA_BUNDLE}")
  message(FATAL_ERROR "Missing CA bundle: ${UPDATE_CA_BUNDLE}")
endif()
add_custom_command(TARGET 1bot-control POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${UPDATE_RUNTIME_FILES} "$<TARGET_FILE_DIR:1bot-control>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${UPDATE_CA_BUNDLE}" "$<TARGET_FILE_DIR:1bot-control>/curl-ca-bundle.crt")
install(FILES ${UPDATE_RUNTIME_FILES} DESTINATION .)
install(FILES "${UPDATE_CA_BUNDLE}" DESTINATION . RENAME curl-ca-bundle.crt)
target_compile_definitions(1bot-control PRIVATE APP_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(1bot-control PRIVATE /utf-8 /W4)
  target_link_libraries(1bot-control PRIVATE Crypt32)
endif()
add_executable(1bot-updater updater_main.cpp)
target_link_libraries(1bot-updater PRIVATE Qt5::Core)
if(MSVC)
  target_compile_options(1bot-updater PRIVATE /utf-8 /W4)
endif()
install(TARGETS 1bot-control 1bot-updater RUNTIME DESTINATION .)
